model Collector {
  id                     Int                      @id @default(autoincrement())
  name                   String
  role                   String
  groupType              String
  position               Int
  active                 Boolean                  @default(true)
  createdAt              DateTime                 @default(now())
  updatedAt              DateTime                 @updatedAt
  bio                    String?
  funFact                String?
  isEmployeeOfMonth      Boolean                  @default(false)
  photoUrl               String?
  preferredName          String?
  profileTitle           String?
  recognitionMessage     String?
  showOnMeetTheBees      Boolean                  @default(true)
  hourlyEntries          HourlyOperationalEntry[]
  leadForagerReigns      LeadForagerReign[]
  monthlyLeadForagerWins MonthlyLeadForager[]
  emfEntries             QualityEmfEntry[]
  stickEntries           WorkerStickEntry[]

  @@index([active])
  @@index([position])
  @@index([showOnMeetTheBees])
  @@index([isEmployeeOfMonth])
}

model WorkerStickEntry {
  id               Int       @id @default(autoincrement())
  collectorId      Int
  entryDate        DateTime
  totalSticks      Int       @default(0)
  successfulSticks Int       @default(0)
  createdAt        DateTime  @default(now())
  updatedAt        DateTime  @updatedAt
  collector        Collector @relation(fields: [collectorId], references: [id], onDelete: Cascade)

  @@unique([collectorId, entryDate])
  @@index([entryDate])
  @@index([collectorId])
  @@index([collectorId, entryDate])
}

model QualityEmfEntry {

  collectorId Int
  entryDate   DateTime
  emfCount    Int       @default(1)
  note        String?
  createdAt   DateTime  @default(now())

  collector   Collector @relation(fields: [collectorId], references: [id], onDelete: Cascade)

  @@index([collectorId])
  @@index([entryDate])
  @@index([collectorId, entryDate])
}

model DashboardMetric {
  id            Int             @id @default(autoincrement())
  key           String          @unique
  displayName   String
  description   String?
  unit          String?
  decimalPlaces Int             @default(2)
  publicSource  MetricSource    @default(CSL)
  isVisible     Boolean         @default(true)
  displayOrder  Int             @default(0)
  createdAt     DateTime        @default(now())
  updatedAt     DateTime        @updatedAt
  readings      MetricReading[]

  @@index([isVisible, displayOrder])
}

model MetricReading {
  id         Int             @id @default(autoincrement())
  metricId   Int
  source     MetricSource
  value      Float
  recordedAt DateTime        @default(now())
  createdAt  DateTime        @default(now())
  updatedAt  DateTime        @updatedAt
  metric     DashboardMetric @relation(fields: [metricId], references: [id], onDelete: Cascade)

  @@index([metricId, source, recordedAt])
  @@index([recordedAt])
}

enum MetricSource {
  CSL
  HIVE
  MANUAL
}
